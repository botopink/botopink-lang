fn validate(items: i32) -> Result<i32, string> {
    for (0..items) { i ->
        if (i > 2) { throw "too many"; };
    };
    return items;
}
fn main() -> void {
    print(validate(2).isOk());
}

import {io.terminal.print} from "std";
