fn average(xs: array<f64>) -> f64 {
    var total = 0.0;
    var n = 0.0;
    for (xs) { x ->
        total = total + x;
        n = n + 1.0;
    };
    return total / n;
}

    val cat: fn(string, string) -> string = { x, y -> x + y };
